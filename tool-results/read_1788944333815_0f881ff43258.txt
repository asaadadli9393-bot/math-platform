   179→function formatDate(s: string | null): string {
   180→  if (!s) return "—";
   181→  try {
   182→    return new Date(s).toLocaleString("ar-DZ", {
   183→      year: "numeric",
   184→      month: "2-digit",
   185→      day: "2-digit",
   186→      hour: "2-digit",
   187→      minute: "2-digit",
   188→    });
   189→  } catch {
   190→    return s;
   191→  }
   192→}
   193→
   194→function formatMoney(n: number): string {
   195→  return new Intl.NumberFormat("fr-DZ").format(n) + " دج";
   196→}

   198→function buildAuthHeaders(token: string, json = false): HeadersInit {
   199→  const h: Record<string, string> = {
   200→    Authorization: `Bearer ${token}`,
   201→  };
   202→  if (json) h["Content-Type"] = "application/json";
   203→  return h as HeadersInit;
   204→}

   210→export default function AdminPage() {
   211→  const [token, setToken] = React.useState<string | null>(null);
   212→
   213→  // قراءة الرمز من sessionStorage بعد التحميل
   214→  React.useEffect(() => {

   217→      if (saved) setToken(saved);
   218→    } catch {
   219→      // تجاهل
   220→    }
   221→  }, []);
   222→
   223→  const handleLogin = (t: string) => {
   224→    setToken(t);
   225→    try {
   226→      sessionStorage.setItem("admin-token", t);
   227→    } catch {
   228→      // تجاهل
   229→    }
   230→  };
   231→
   232→  const handleLogout = () => {
   233→    setToken(null);
   234→    try {
   235→      sessionStorage.removeItem("admin-token");
   236→    } catch {
   237→      // تجاهل
   238→    }
   239→  };
   240→
   241→  if (!token) {
   242→    return <LoginScreen onLogin={handleLogin} />;
   243→  }
   244→  return <AdminDashboard token={token} onLogout={handleLogout} />;

   450→function PaymentsTab({ token }: { token: string }) {
   451→  const [payments, setPayments] = React.useState<PaymentRow[]>([]);
   452→  const [loading, setLoading] = React.useState(false);
   453→  const [error, setError] = React.useState<string | null>(null);
   454→  const [filterStatus, setFilterStatus] = React.useState<string>("ALL");
   455→  const [filterPlan, setFilterPlan] = React.useState<string>("ALL");
   456→  const [search, setSearch] = React.useState("");
   457→  const [actingId, setActingId] = React.useState<string | null>(null);
   458→  const { toast } = useToast();
   459→
   460→  const load = React.useCallback(async () => {
   461→    setLoading(true);
   462→    setError(null);
   463→    try {
   464→      const params = new URLSearchParams();
   465→      if (filterStatus !== "ALL") params.set("status", filterStatus);
   466→      if (filterPlan !== "ALL") params.set("plan", filterPlan);
   467→      const res = await fetch(
   468→        `/api/admin/payments?${params.toString()}`,
   469→        { headers: buildAuthHeaders(token) }
   470→      );
   471→      const data = await res.json();
   472→      if (!res.ok) {
   473→        setError(data.error || "تعذّر جلب المدفوعات");
   474→        setPayments([]);
   475→        return;
   476→      }
   477→      setPayments(Array.isArray(data.payments) ? data.payments : []);
   478→    } catch {
   479→      setError("تعذّر الاتصال بالخادم");
   480→    } finally {
   481→      setLoading(false);
   482→    }
   483→  }, [token, filterStatus, filterPlan]);
   484→
   485→  React.useEffect(() => {
   486→    load();
   487→  }, [load]);
   488→
   489→  async function doAction(paymentId: string, action: "verify" | "reject" | "expire" | "cancel") {
   490→    setActingId(paymentId);
   491→    try {
   492→      const res = await fetch("/api/admin/payments", {
   493→        method: "PATCH",
   494→        headers: buildAuthHeaders(token, true),
   495→        body: JSON.stringify({ paymentId, action }),
   496→      });
   497→      const data = await res.json();
   498→      if (!res.ok || !data.success) {
   499→        toast({
   500→          title: "فشل التحديث",
   501→          description: data.error || "تعذّر تحديث الدفعة",
   502→          variant: "destructive",
   503→        });
   504→        return;
   505→      }
   506→      toast({
   507→        title: "تم التحديث",
   508→        description: `الدفعة ${paymentId} — ${action}`,
   509→      });
   510→      await load();
   511→    } finally {
   512→      setActingId(null);
   513→    }
   514→  }
   515→
   516→  // فلترة البحث (محلية)
   517→  const filtered = React.useMemo(() => {
   518→    if (!search.trim()) return payments;
   519→    const q = search.trim().toLowerCase();
   520→    return payments.filter((p) =>
   521→      [p.paymentId, p.phone, p.studentName, p.transactionRef, p.email]
   522→        .filter(Boolean)
   523→        .some((v) => String(v).toLowerCase().includes(q))
   524→    );
   525→  }, [payments, search]);
   526→
   527→  return (
   528→    <div className="space-y-6">
   529→      {/* شريط الفلترة */}
   530→      <Card>
   531→        <CardHeader>
   532→          <CardTitle className="flex items-center gap-2">
   533→            <CreditCard className="w-5 h-5 text-emerald-600" />
   534→            المدفوعات
   535→          </CardTitle>
   536→          <CardDescription>
   537→            مراجعة الدفعات والتحقق منها أو رفضها. الفلترة حسب الحالة والباقة والبحث النصّي.
   538→          </CardDescription>
   539→        </CardHeader>
   540→        <CardContent>
   541→          <div className="flex flex-col md:flex-row md:items-end gap-4">
   542→            <div className="flex-1 space-y-2">
   543→              <Label htmlFor="search">بحث (paymentId / هاتف / رقم العملية)</Label>
   544→              <div className="relative">
   545→                <Search className="w-4 h-4 absolute right-3 top-1/2 -translate-y-1/2 text-muted-foreground" />
   546→                <Input
   547→                  id="search"
   548→                  value={search}
   549→                  onChange={(e) => setSearch(e.target.value)}
   550→                  placeholder="PAY-... أو 06... أو رقم العملية"
   551→                  className="pr-9"
   552→                />
   553→              </div>
   554→            </div>
   555→            <div className="space-y-2">
   556→              <Label>الحالة</Label>
   557→              <Select value={filterStatus} onValueChange={setFilterStatus}>
   558→                <SelectTrigger className="w-full md:w-44">
   559→                  <SelectValue />
   560→                </SelectTrigger>
   561→                <SelectContent>
   562→                  <SelectItem value="ALL">كل الحالات</SelectItem>
   563→                  <SelectItem value="PENDING">قيد المراجعة</SelectItem>
   564→                  <SelectItem value="VERIFIED">موثّقة</SelectItem>
   565→                  <SelectItem value="REJECTED">مرفوضة</SelectItem>
   566→                  <SelectItem value="EXPIRED">منتهية</SelectItem>
   567→                  <SelectItem value="CANCELLED">ملغاة</SelectItem>
   568→                </SelectContent>
   569→              </Select>
   570→            </div>
   571→            <div className="space-y-2">
   572→              <Label>الباقة</Label>
