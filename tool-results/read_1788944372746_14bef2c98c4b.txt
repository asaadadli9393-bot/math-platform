   450→   450→function PaymentsTab({ token }: { token: string }) {
   451→   451→  const [payments, setPayments] = React.useState<PaymentRow[]>([]);
   452→   452→  const [loading, setLoading] = React.useState(false);
   453→   453→  const [error, setError] = React.useState<string | null>(null);
   454→   454→  const [filterStatus, setFilterStatus] = React.useState<string>("ALL");
   455→   455→  const [filterPlan, setFilterPlan] = React.useState<string>("ALL");
   456→   456→  const [search, setSearch] = React.useState("");
   457→   457→  const [actingId, setActingId] = React.useState<string | null>(null);
   458→   458→  const { toast } = useToast();
   459→   459→
   460→   460→  const load = React.useCallback(async () => {
   461→   461→    setLoading(true);
   462→   462→    setError(null);
   463→   463→    try {
   464→   464→      const params = new URLSearchParams();
   465→   465→      if (filterStatus !== "ALL") params.set("status", filterStatus);
   466→   466→      if (filterPlan !== "ALL") params.set("plan", filterPlan);
   467→   467→      const res = await fetch(
   468→   468→        `/api/admin/payments?${params.toString()}`,
   469→   469→        { headers: buildAuthHeaders(token) }
   470→   470→      );
   471→   471→      const data = await res.json();
   472→   472→      if (!res.ok) {
   473→   473→        setError(data.error || "تعذّر جلب المدفوعات");
   474→   474→        setPayments([]);
   475→   475→        return;
   476→   476→      }
   477→   477→      setPayments(Array.isArray(data.payments) ? data.payments : []);
   478→   478→    } catch {
   479→   479→      setError("تعذّر الاتصال بالخادم");
