    72→function formatBytes(b: number): string {
    73→  if (b < 1024) return `${b} B`;
    74→  if (b < 1024 * 1024) return `${(b / 1024).toFixed(1)} KB`;
    75→  if (b < 1024 * 1024 * 1024) return `${(b / 1024 / 1024).toFixed(1)} MB`;
    76→  return `${(b / 1024 / 1024 / 1024).toFixed(2)} GB`;
    77→}
    78→
    79→function formatDate(s: string): string {
    80→  try {
    81→    return new Date(s).toLocaleString("ar-DZ", {
    82→      year: "numeric",
    83→      month: "2-digit",
    84→      day: "2-digit",
    85→      hour: "2-digit",
    86→      minute: "2-digit",
    87→    });
    88→  } catch {
    89→    return s;
    90→  }
    91→}
